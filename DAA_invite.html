//This is for DB CREATE DATABASE IF NOT EXISTS library_db; USE library_db; CREATE TABLE IF NOT EXISTS books ( id INT PRIMARY KEY, title VARCHAR(150) NOT NULL, author VARCHAR(100) NOT NULL, category VARCHAR(80) NOT NULL, isbn VARCHAR(30) NOT NULL, copies INT NOT NULL ); //server.js const http = require("http"); const fs = require("fs"); const path = require("path"); const mysql = require("mysql2"); const PORT = 3000; // Connect to XAMPP MySQL const db = mysql.createConnection({ host: "localhost", user: "root", password: "", database: "library_db" }); db.connect(err => { if (err) { console.log("Database connection failed:", err.message); return; } console.log("MySQL connected successfully!"); }); function sendJSON(res, status, data) { res.writeHead(status, { "Content-Type": "application/json" }); res.end(JSON.stringify(data)); } function readBody(req, callback) { let body = ""; req.on("data", chunk => body += chunk); req.on("end", () => { try { callback(null, JSON.parse(body || "{}")); } catch { callback(new Error("Invalid JSON")); } }); } const server = http.createServer((req, res) => { const url = new URL(req.url, "http://localhost"); // READ if (req.method === "GET" && url.pathname === "/books") { db.query("SELECT * FROM books", (err, rows) => { if (err) return sendJSON(res, 500, { error: err.message }); sendJSON(res, 200, rows); }); } // CREATE else if (req.method === "POST" && url.pathname === "/books") { readBody(req, (err, book) => { if (err) return sendJSON(res, 400, { error: err.message }); const sql = `INSERT INTO books (id, title, author, category, isbn, copies) VALUES (?, ?, ?, ?, ?, ?)`; const values = [ book.id, book.title, book.author, book.category, book.isbn, book.copies ]; db.query(sql, values, err => { if (err) return sendJSON(res, 500, { error: err.message }); sendJSON(res, 201, { message: "Book added successfully" }); }); }); } // UPDATE else if (req.method === "PUT" && url.pathname.startsWith("/books/")) { const id = url.pathname.split("/")[2]; readBody(req, (err, book) => { if (err) return sendJSON(res, 400, { error: err.message }); const sql = `UPDATE books SET title = ?, author = ?, category = ?, isbn = ?, copies = ? WHERE id = ?`; const values = [ book.title, book.author, book.category, book.isbn, book.copies, id ]; db.query(sql, values, (err, result) => { if (err) return sendJSON(res, 500, { error: err.message }); if (result.affectedRows === 0) { return sendJSON(res, 404, { message: "Book ID not found" }); } sendJSON(res, 200, { message: "Book updated successfully" }); }); }); } // DELETE else if (req.method === "DELETE" && url.pathname.startsWith("/books/")) { const id = url.pathname.split("/")[2]; db.query("DELETE FROM books WHERE id = ?", [id], (err, result) => { if (err) return sendJSON(res, 500, { error: err.message }); if (result.affectedRows === 0) { return sendJSON(res, 404, { message: "Book ID not found" }); } sendJSON(res, 200, { message: "Book deleted successfully" }); }); } // Serve Bootstrap frontend else if (req.method === "GET" && url.pathname === "/") { const file = path.join(__dirname, "public", "index.html"); fs.readFile(file, (err, content) => { if (err) { res.writeHead(500, { "Content-Type": "text/plain" }); return res.end("Cannot load index.html"); } res.writeHead(200, { "Content-Type": "text/html" }); res.end(content); }); } else { sendJSON(res, 404, { message: "Route not found" }); } }); server.listen(PORT, () => { console.log(`Server running at http://localhost:${PORT}`); }); //public/index.html Library Management
